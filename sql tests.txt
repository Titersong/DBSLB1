SELECT NAME, ISBN, PUBLISHER, YEAR, NUMBER, QUANTITY
FROM literature
WHERE PUBLISHER = '3A Corporation'
AND LITERATE = 'Book';

SELECT NAME, PUBLISHER, YEAR, LITERATE, ISBN, NUMBER, QUANTITY
FROM literature
WHERE YEAR BETWEEN 2000 AND 2025
AND LITERATE IN ('Book', 'Journal', 'Newspaper');

SELECT l.NAME, l.PUBLISHER, l.YEAR
FROM author a
JOIN book_authrs ba ON a.Id = ba.FID_AUTH
JOIN literature l ON l.Id = ba.FID_BOOK
WHERE a.Id = 10;
